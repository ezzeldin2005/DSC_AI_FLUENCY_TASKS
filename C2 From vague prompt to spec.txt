```
## 1. Endpoint: `PATCH /api/courier/orders/{orderId}/deliver`
Here is a sample implementation using a controller, EF Core, and ASP.NET Core authorization.

```
using System.Security.Claims;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

[ApiController]
[Route("api/courier/orders")]
[Authorize(Roles = "Courier")]
public class CourierOrdersController : ControllerBase
{
    private readonly AppDbContext _context;

    public CourierOrdersController(AppDbContext context){_context = context;}

    [HttpPatch("{orderId:int}/deliver")]
    public async Task<IActionResult> MarkAsDelivered(int orderId){
        
        // 1. Get the authenticated courier's ID
        var courierId = User.FindFirstValue(ClaimTypes.NameIdentifier);
        if (courierId == null)return Unauthorized();
        
        // 2. Find the order
        var order = await _context.Orders.FirstOrDefaultAsync(o => o.Id == orderId);

        if (order == null)return NotFound(new { message = "Order not found." });

        // 3. Ensure this courier is assigned to the order
        if (order.CourierId != courierId)return Forbid();

        // 4. Validate the current order status
        if (order.Status != OrderStatus.OutForDelivery){return Conflict(new{message = "Only orders out for delivery can be marked as delivered."});}

        // 5. Update the order
        order.Status = OrderStatus.Delivered;
        order.DeliveredAt = DateTime.UtcNow;

        await _context.SaveChangesAsync();

        // 6. Return the updated result
        return Ok(new{orderId = order.Id,status = order.Status.ToString(),deliveredAt = order.DeliveredAt});
    }
}

```
## 3. Example request and response
Request
```
PATCH /api/courier/orders/42/deliver
Authorization: Bearer <courier-token>
```
No request body is needed because the server determines the delivery time and the courier identity from the authenticated user's claims.

Successful response — `200 OK`
```
{"orderId": 42,"status": "Delivered","deliveredAt": "2026-10-17T08:10:00Z"}
```
Other possible responses:

| Status code        | Meaning                                                              |
| ------------------ | -------------------------------------------------------------------- |
| `401 Unauthorized` | User is not authenticated.                                           |
| `403 Forbidden`    | User is not authorized for the order, or is not in the Courier role. |
| `404 Not Found`    | Order does not exist.                                                |
| `409 Conflict`     | Order is not currently out for delivery.                             |

## 4. Important production considerations
- Authorization: Never accept `CourierId` from the request body to establish the caller's identity.
- Concurrency: Two simultaneous requests could both pass the status check. Consider EF Core concurrency tokens or a conditional database update.
- Business rules: If your workflow permits delivery only after payment confirmation or requires proof of delivery, enforce those rules before updating the order.
- Architecture: In a layered .NET application, keep the controller thin and move the validation and delivery operation into an application service or BLL.

Design choice: `PATCH` is appropriate because the endpoint changes only the order's delivery state rather than replacing the entire order resource.
```

```
// CourierOrdersController.cs

using System.Security.Claims;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

[ApiController]
[Route("api/courier/orders")]
[Authorize(Roles = "Courier")]
public class CourierOrdersController : ControllerBase
{
    private readonly AppDbContext _db;

    public CourierOrdersController(AppDbContext db){_db = db;}

    [HttpPost("{orderId:int}/deliver")]
    public async Task<IActionResult> Deliver(int orderId,[FromBody] DeliverOrderRequest request)
    {
        // Assumption: authentication middleware validates the JWT and
        // populates ClaimTypes.NameIdentifier and the Courier role.
        var courierId = User.FindFirstValue(ClaimTypes.NameIdentifier);

        if (string.IsNullOrWhiteSpace(courierId))return Unauthorized();

        if (request is null ||string.IsNullOrWhiteSpace(request.ProofValue)){return BadRequest(new { message = "ProofValue is required." });}

        var order = await _db.Orders.FirstOrDefaultAsync(o => o.Id == orderId);

        if (order is null)return NotFound();

        if (!string.Equals(order.CourierId, courierId, StringComparison.Ordinal))return StatusCode(StatusCodes.Status403Forbidden);

        if (order.Status != "OutForDelivery")return Conflict(new{message = "Only OutForDelivery orders can be delivered."});

        if (order.PaymentMethod == "COD"){
            if (!request.CollectedAmount.HasValue ||request.CollectedAmount.Value != order.Total){return BadRequest(new{message = "CollectedAmount must equal the order total for COD."});}
        }
        else if (order.PaymentMethod == "Prepaid"){
            if (request.CollectedAmount.HasValue){return BadRequest(new{message = "CollectedAmount must be absent for prepaid orders."});}
        }
        else{
            // Assumption: unsupported payment methods are invalid requests.
            return BadRequest(new { message = "Unsupported payment method." });
        }

        // Defense in depth: do not overwrite an existing delivery timestamp.
        if (order.DeliveredAt.HasValue){return Conflict(new{message = "The order has already been delivered."});}

        order.ProofValue = request.ProofValue.Trim();
        order.CollectedAmount = request.CollectedAmount;
        order.DeliveredAt = DateTime.UtcNow;
        order.Status = "Delivered";

        try{ await _db.SaveChangesAsync();}
        catch (DbUpdateConcurrencyException){return Conflict(new{message = "The order was modified concurrently. Refresh and retry."});}

        return Ok(new{orderId = order.Id,status = order.Status,deliveredAt = order.DeliveredAt.Value});
    }
}
```
